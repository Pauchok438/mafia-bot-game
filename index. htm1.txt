    <script>
        let currentRole = null;
        let gameActive = false;
        let phase = "day";
        
        const roles = {
            mafia: {
                name: "Мафия",
                icon: "🔪",
                desc: "Ты — член мафии. Ночью ты убиваешь мирных жителей. Твоя цель — остаться последним."
            },
            civilian: {
                name: "Мирный житель",
                icon: "🏠",
                desc: "Ты — мирный житель. Днём ты голосуешь против подозреваемых. Твоя цель — найти мафию."
            },
            commissioner: {
                name: "Комиссар",
                icon: "🔍",
                desc: "Ты — комиссар полиции. Ночью можешь проверять, мафия ли игрок."
            },
            doctor: {
                name: "Доктор",
                icon: "💊",
                desc: "Ты — доктор. Ночью можешь лечить одного игрока, спасая его от убийства."
            }
        };
        
        function addLog(message, type = "info") {
            const logDiv = document.getElementById("eventLog");
            const p = document.createElement("p");
            p.innerHTML = message;
            if (type === "night") p.classList.add("night");
            if (type === "day") p.classList.add("day");
            logDiv.appendChild(p);
            logDiv.scrollTop = logDiv.scrollHeight;
            while (logDiv.children.length > 15) {
                logDiv.removeChild(logDiv.firstChild);
            }
        }
        
        function updateUI() {
            if (!currentRole) {
                document.getElementById("roleIcon").innerText = "❓";
                document.getElementById("roleName").innerText = "Неизвестно";
                document.getElementById("roleDesc").innerText = "Нажмите «Новая игра», чтобы получить роль";
                return;
            }
            const roleData = roles[currentRole];
            document.getElementById("roleIcon").innerText = roleData.icon;
            document.getElementById("roleName").innerText = roleData.name;
            document.getElementById("roleDesc").innerText = roleData.desc;
        }
        
        function newGame() {
            const roleKeys = ["mafia", "civilian", "commissioner", "doctor"];
            const randomIndex = Math.floor(Math.random() * roleKeys.length);
            currentRole = roleKeys[randomIndex];
            gameActive = true;
            phase = "day";
            updateUI();
            addLog("✨ Новая игра началась! ✨", "day");
            addLog(`📜 Вы получили роль: ${roles[currentRole].name} ${roles[currentRole].icon}`);
            addLog("🌞 Наступил день. Город просыпается.", "day");
            
            if (currentRole === "mafia") {
                addLog("💀 Ночью вы сможете убить одного игрока. Используйте силу с умом.", "night");
            } else if (currentRole === "commissioner") {
                addLog("🔍 Ночью вы сможете проверить, мафия ли игрок.", "night");
            } else if (currentRole === "doctor") {
                addLog("💊 Ночью вы сможете спасти игрока от смерти.", "night");
            } else {
                addLog("🌙 Ночью вы спите. Будьте внимательны днём.", "night");
            }
        }
        
        function nightAction() {
            if (!gameActive) {
                addLog("⚠️ Сначала начните новую игру!", "info");
                return;
            }
            if (phase !== "night") {
                addLog("🌞 Сейчас день! Сначала проведите голосование, чтобы наступила ночь.", "day");
                return;
            }
            
            if (currentRole === "mafia") {
                addLog("🔪 Вы выбрали жертву... Мирный житель убит этой ночью.", "night");
                addLog("💀 Кто-то не проснулся этим утром.", "night");
            } else if (currentRole === "commissioner") {
                const randomCheck = Math.random() > 0.6;
                if (randomCheck) {
                    addLog("🔍 Вы провели расследование. Этот человек — МАФИЯ!", "night");
                } else {
                    addLog("🔍 Вы проверили подозреваемого. Он чист. Мирный житель.", "night");
                }
            } else if (currentRole === "doctor") {
                addLog("💊 Вы спасли игрока этой ночью. Кто-то остался жив благодаря вам.", "night");
            } else {
                addLog("🌙 Вы мирно спали и ничего не видели...", "night");
            }
            
            phase = "day";
            addLog("🌞 Наступило утро. Город просыпается.", "day");
            addLog("🗳️ Обсудите события ночи и проголосуйте за подозреваемого.", "day");
        }
        
        function dayVote() {
            if (!gameActive) {
                addLog("⚠️ Сначала начните новую игру!", "info");
                return;
            }
            if (phase !== "day") {
                addLog("🌙 Сейчас ночь! Сначала совершите ночное действие.", "night");
                return;
            }
            
            const randomOutcome = Math.random();
            
            if (currentRole === "mafia" && randomOutcome > 0.5) {
                addLog("🗳️ Голосование окончено. Город казнил МИРНОГО ЖИТЕЛЯ! Мафия торжествует.", "day");
                addLog("💀 Мафия одержала победу. Начните новую игру.", "day");
                gameActive = false;
            } else if (currentRole === "mafia" && randomOutcome <= 0.5) {
                addLog("🗳️ Голосование окончено. Город казнил ПОДОЗРЕВАЕМОГО. Это оказалась МАФИЯ!", "day");
                addLog("🏆 Мирные жители празднуют победу!", "day");
                gameActive = false;
            } else if (currentRole !== "mafia" && randomOutcome > 0.4) {
                addLog("🗳️ Голосование окончено. Город казнил МАФИЮ! Правосудие восторжествовало.", "day");
                addLog("🏆 Мирные жители празднуют победу!", "day");
                gameActive = false;
            } else {
                addLog("🗳️ Голосование окончено. Город казнил МИРНОГО ЖИТЕЛЯ. Мафия остаётся в тени.", "day");
                if (currentRole !== "mafia") {
                    addLog("💀 Вы были казнены по ошибке... Игра окончена.", "day");
                    gameActive = false;
                } else {
                    addLog("🔪 Мафия продолжает своё дело. Наступает ночь.", "day");
                    phase = "night";
                    addLog("🌙 Город засыпает. Наступает время ночных действий.", "night");
                }
            }
            
            if (!gameActive) {
                addLog("🔄 Нажмите «Новая игра», чтобы сыграть снова.", "info");
            }
        }
        
        document.getElementById("newGameBtn").addEventListener("click", newGame);
        document.getElementById("nightActionBtn").addEventListener("click", nightAction);
        document.getElementById("voteBtn").addEventListener("click", dayVote);
        
        newGame();
    </script>
